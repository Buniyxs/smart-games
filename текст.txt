        function setupPeerConnection(conn) {
            let assignedSlot = players.findIndex(p => !p.connected);
            if (assignedSlot === -1 && players.length < 4) {
                assignedSlot = players.length;
                addPlayer(assignedSlot);
            }
            if (assignedSlot !== -1) {
                players[assignedSlot].connected = true;
                conn.send({ type: 'init', playerIndex: assignedSlot, classData: PLAYER_CLASSES[assignedSlot] });
            }

            conn.on('data', (data) => handleRemoteControllerInput(data));
            conn.on('close', () => {
                if (assignedSlot !== -1 && players[assignedSlot]) players[assignedSlot].connected = false;
            });
            document.getElementById('tv-status').innerText = `${players.filter(p => p.connected).length} Joystik ulandi!`;
        }

        function connectToTV() {
            const inputCode = document.getElementById('room-code-input').value.trim();
            if (inputCode.length !== 4) return;
            roomCode = inputCode;
            document.getElementById('status-text').innerText = 'Ulanmoqda...';

            broadcastChannel = new BroadcastChannel('zombie_arena_' + roomCode);
            broadcastChannel.onmessage = (e) => {
                if (e.data.type === 'sync') updateJoystickUISync(e.data);
            };

            try {
                peer = new Peer();
                peer.on('open', () => {
                    const conn = peer.connect('zombie-tv-' + roomCode);
                    conn.on('open', () => {
                        peerConn = conn;
                        conn.on('data', (data) => {
                            if (data.type === 'init') setupJoystickPlayerBadge(data.playerIndex, data.classData);
                            else if (data.type === 'sync') updateJoystickUISync(data);
                        });
                        onJoystickConnected();
                    });
                });
            } catch (err) {}

            setTimeout(() => {
                onJoystickConnected();
                broadcastChannel.postMessage({ type: 'join' });
            }, 500);
        }

        function sendJoystickInput(inputPacket) {
            if (navigator.vibrate && inputPacket.shoot) navigator.vibrate(8);
            const data = { type: 'input', peerId: peer ? peer.id : 'local', ...inputPacket };
            if (broadcastChannel) broadcastChannel.postMessage(data);
            if (peerConn && peerConn.open) peerConn.send(data);
        }

        function handleRemoteControllerInput(data) {
            if (data.type === 'join') {
                if (players.length < 4) addPlayer(players.length);
                return;
            }
            if (data.type === 'input') {
                const targetP = players[data.playerIndex || 0];
                if (targetP) {
                    targetP.inputLeft = data.left;
                    targetP.inputRight = data.right;
                    targetP.inputShoot = data.shoot;
                    if (data.reload) targetP.reloadWeapon();
                    if (data.ability) targetP.useAbility();
                }
            }
        }

        function syncStateToControllers() {
            if (currentMode !== 'tv' || (!broadcastChannel && !peerConn)) return;
            players.forEach((p, idx) => {
                const packet = {
                    type: 'sync',
                    playerIndex: idx,

                    maxHp: p.maxHp,
                    weaponName: p.weapon.name,
                    weaponIcon: p.weapon.icon,
                    ammo: p.ammo,

                    abilityCd: Math.ceil(p.abilityCooldown)
                };
                if (broadcastChannel) broadcastChannel.postMessage(packet);
                if (peerConn && peerConn.open) peerConn.send(packet);
            });
        }

        function setupTouchControlBindings(leftId, rightId, shootId, reloadId, abilityId) {
            const inputState = { left: false, right: false, shoot: false, reload: false, ability: false };

            const bindBtn = (elId, key) => {
                const el = document.getElementById(elId);
                if (!el) return;

                const down = (e) => {
                    e.preventDefault();
                    initAudio();
                    el.classList.add('active');

                    if (key === 'reload') {
                        if (currentMode === 'single' && players[0]) players[0].reloadWeapon();
                        else sendJoystickInput({ ...inputState, reload: true });
                    } else if (key === 'ability') {
                        if (currentMode === 'single' && players[0]) players[0].useAbility();
                        else sendJoystickInput({ ...inputState, ability: true });
                    } else {
                        inputState[key] = true;
                        sendCurrentInput();
                    }
                };

                const up = (e) => {
                    e.preventDefault();
                    el.classList.remove('active');
                    if (key !== 'reload' && key !== 'ability') {
                        inputState[key] = false;
                        sendCurrentInput();
                    }
                };

                el.addEventListener('touchstart', down, { passive: false });
                el.addEventListener('touchend', up, { passive: false });
                el.addEventListener('mousedown', down);
                el.addEventListener('mouseup', up);
            };

            bindBtn(leftId, 'left');
            bindBtn(rightId, 'right');
            bindBtn(shootId, 'shoot');
            bindBtn(reloadId, 'reload');
            bindBtn(abilityId, 'ability');

            function sendCurrentInput() {
                if (currentMode === 'single' && players[0]) {
                    players[0].inputLeft = inputState.left;
                    players[0].inputRight = inputState.right;
                    players[0].inputShoot = inputState.shoot;
                } else {
                    sendJoystickInput(inputState);
                }
            }
        }

        function initAudio() {
            if (audioCtx) return;
            try { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } catch(e){}
        }

        function playSound(type) {
            if (!audioCtx) return;
            try {
                const now = audioCtx.currentTime;
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.connect(gain);
                gain.connect(audioCtx.destination);

                if (type === 'shoot') {
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(420, now);
                    osc.frequency.exponentialRampToValueAtTime(30, now + 0.12);
                    gain.gain.setValueAtTime(0.25, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.12);
                    osc.start(now); osc.stop(now + 0.12);
                } else if (type === 'plasma') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(950, now);
                    osc.frequency.exponentialRampToValueAtTime(100, now + 0.25);
                    gain.gain.setValueAtTime(0.3, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.25);
                    osc.start(now); osc.stop(now + 0.25);
                } else if (type === 'explosion') {
                    osc.type = 'square';
                    osc.frequency.setValueAtTime(140, now);
                    osc.frequency.exponentialRampToValueAtTime(15, now + 0.45);
                    gain.gain.setValueAtTime(0.4, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.45);
                    osc.start(now); osc.stop(now + 0.45);
                } else if (type === 'reload') {
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(500, now);
                    osc.frequency.setValueAtTime(800, now + 0.08);
                    gain.gain.setValueAtTime(0.12, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.15);
                    osc.start(now); osc.stop(now + 0.15);
                } else if (type === 'zombie') {
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(90, now);
                    osc.frequency.linearRampToValueAtTime(140, now + 0.2);
                    osc.frequency.linearRampToValueAtTime(50, now + 0.35);
                    gain.gain.setValueAtTime(0.15, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.35);
                    osc.start(now); osc.stop(now + 0.35);
                } else if (type === 'upgrade') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(400, now);
                    osc.frequency.setValueAtTime(650, now + 0.1);
                    osc.frequency.setValueAtTime(950, now + 0.2);
                    gain.gain.setValueAtTime(0.3, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.35);
                    osc.start(now); osc.stop(now + 0.35);
                }
            } catch(e) {}
        }

        class Player {
            constructor(slot, classIndex = slot) {
                this.slot = slot;
                this.classData = PLAYER_CLASSES[classIndex];
                this.x = 220 + slot * 140;
                this.y = groundY;
                this.vx = 0;
                this.speed = 4.4;
                this.hp = 100;
                this.maxHp = 100;
                this.facing = 1;
                this.kills = 0;
                this.weaponIndex = 0;
                this.weapon = WEAPONS[0];
                this.ammo = this.weapon.ammoMax;
                this.isReloading = false;
                this.reloadTimer = 0;
                this.shootCooldown = 0;
                this.abilityCooldown = 0;
                this.infiniteAmmoTimer = 0;
                this.connected = true;
                this.isDead = false;
                this.animFrame = 0;

                this.inputLeft = false;
                this.inputRight = false;
                this.inputShoot = false;
            }

            update(dt) {
                if (this.isDead) return;

                this.vx = 0;
                if (this.inputLeft) { this.vx = -this.speed; this.facing = -1; }
                if (this.inputRight) { this.vx = this.speed; this.facing = 1; }

                this.x += this.vx;
                this.x = Math.max(40, Math.min(canvas.width - 40, this.x));

                if (Math.abs(this.vx) > 0) this.animFrame += 0.22;

                if (this.shootCooldown > 0) this.shootCooldown -= dt;
                if (this.infiniteAmmoTimer > 0) this.infiniteAmmoTimer -= dt;

                if (this.isReloading) {
                    this.reloadTimer -= dt;
                    if (this.reloadTimer <= 0) {
                        this.isReloading = false;
                        this.ammo = this.weapon.ammoMax;
                    }
                }

                if (this.abilityCooldown > 0) {
                    this.abilityCooldown -= dt;
                    if (this.slot === 0 && currentMode === 'single') {
                        const cdEl = document.getElementById('sp-ability-cd');
                        if (cdEl) {
                            if (this.abilityCooldown > 0) {
                                cdEl.classList.remove('hidden');
                                cdEl.innerText = Math.ceil(this.abilityCooldown);
                            } else {
                                cdEl.classList.add('hidden');
                            }
                        }
                    }
                }

                if (this.inputShoot && this.shootCooldown <= 0 && !this.isReloading) {
                    this.shoot();
                }
            }

            shoot() {
                if (this.ammo <= 0 && this.infiniteAmmoTimer <= 0) {
                    this.reloadWeapon();
                    return;
                }

                const effectiveRate = this.infiniteAmmoTimer > 0 ? this.weapon.fireRate * 0.55 : this.weapon.fireRate;
                this.shootCooldown = effectiveRate;

                if (this.infiniteAmmoTimer <= 0) this.ammo--;

                playSound(this.weapon.piercing ? 'plasma' : 'shoot');
                triggerScreenShake(this.weapon.shake, 0.15);

                const barrelX = this.x + this.facing * 28;
                const barrelY = this.y - 34;

                for (let i = 0; i < this.weapon.count; i++) {
                    const angleOffset = (Math.random() - 0.5) * this.weapon.spread;
                    const vx = (this.facing * this.weapon.speed) * Math.cos(angleOffset);
                    const vy = (this.facing * this.weapon.speed) * Math.sin(angleOffset);

                    bullets.push(new Bullet(
                        barrelX, barrelY, vx, vy,
                        this.weapon.damage, this.classData.color,
                        this.weapon.piercing, this.slot
                    ));
                }

                particles.push(new MuzzleFlash(barrelX, barrelY, this.facing, this.weapon.color));
                particles.push(new ShellCasing(barrelX - this.facing * 10, barrelY, -this.facing * (2 + Math.random())));

                if (this.ammo === 0 && !this.isReloading && this.infiniteAmmoTimer <= 0) {
                    this.reloadWeapon();
                }
            }

            reloadWeapon() {
                if (this.isReloading || this.ammo === this.weapon.ammoMax) return;
                this.isReloading = true;
                this.reloadTimer = this.weapon.reloadTime;
                playSound('reload');
            }

            addKill() {
                this.kills++;
                for (let i = WEAPONS.length - 1; i >= 0; i--) {
                    if (this.kills >= WEAPONS[i].minKills && this.weaponIndex < i) {
                        this.weaponIndex = i;
                        this.weapon = WEAPONS[i];
                        this.ammo = this.weapon.ammoMax;
                        this.isReloading = false;
                        playSound('upgrade');
                        floatingTexts.push(new FloatingText(this.x, this.y - 75, `QUROL: ${this.weapon.name}!`, '#facc15', 1.8));
                        break;
                    }
                }
            }

            useAbility() {
                if (this.abilityCooldown > 0 || this.isDead) return;
                this.abilityCooldown = this.classData.cd;

                if (this.classData.id === 0) {
                    this.infiniteAmmoTimer = 10;
                    floatingTexts.push(new FloatingText(this.x, this.y - 65, "CHEKSIZ O'Q!", "#3b82f6", 1.5));
                } else if (this.classData.id === 1) {
                    const barX = this.x + this.facing * 65;
                    barricades.push(new Barricade(barX, groundY));
                    floatingTexts.push(new FloatingText(this.x, this.y - 65, "HIMOYA DEVORI!", "#10b981", 1.5));
                } else if (this.classData.id === 2) {
                    grenades.push(new Grenade(this.x, this.y - 38, this.facing * 13, -9, this.slot));
                    floatingTexts.push(new FloatingText(this.x, this.y - 65, "GRANATA!", "#f59e0b", 1.5));
                } else if (this.classData.id === 3) {
                    lasers.push(new LaserBeam(this.x, this.y - 34, this.facing, this.slot));
                    playSound('plasma');
                    triggerScreenShake(12, 0.4);
                    floatingTexts.push(new FloatingText(this.x, this.y - 65, "LAZER ZARBASI!", "#a855f7", 1.5));
                }
            }

            takeDamage(dmg) {
                if (this.isDead) return;
                this.hp -= dmg;
                floatingTexts.push(new FloatingText(this.x, this.y - 45, `-${Math.round(dmg)}`, '#ef4444', 0.8));
                if (this.hp <= 0) {
                    this.hp = 0;
                    this.isDead = true;
                    floatingTexts.push(new FloatingText(this.x, this.y - 50, "MAG'LUB BO'LDI!", '#dc2626', 2.0));
                }
            }

            draw() {
                if (this.isDead) return;

                ctx.save();
                ctx.translate(this.x, this.y);

                // Laser Sight Beam
                ctx.strokeStyle = 'rgba(239, 68, 68, 0.35)';
                ctx.lineWidth = 1.5;
                ctx.beginPath();
                ctx.moveTo(this.facing * 20, -34);
                ctx.lineTo(this.facing * (canvas.width), -34);
                ctx.stroke();

                if (this.infiniteAmmoTimer > 0) {
                    ctx.strokeStyle = '#3b82f6';
                    ctx.lineWidth = 3;
                    ctx.beginPath();
                    ctx.arc(0, -32, 34 + Math.sin(performance.now() * 0.01) * 4, 0, Math.PI * 2);
                    ctx.stroke();
                }

                ctx.scale(this.facing, 1);

                const legAngle = Math.sin(this.animFrame) * 0.45;
                ctx.fillStyle = '#0f172a';
                ctx.strokeStyle = '#1e293b';
                ctx.lineWidth = 6;

                // Legs
                ctx.beginPath(); ctx.moveTo(-5, -20); ctx.lineTo(-5 + Math.sin(legAngle)*12, 0); ctx.stroke();
                ctx.beginPath(); ctx.moveTo(5, -20); ctx.lineTo(5 - Math.sin(legAngle)*12, 0); ctx.stroke();

                // Tactical Vest & Armor
                ctx.fillStyle = '#1e293b';
                ctx.fillRect(-11, -44, 22, 26);
                ctx.fillStyle = this.classData.color;
                ctx.fillRect(-8, -42, 16, 10);

                // Helmet & Headlamp
                ctx.fillStyle = '#334155';
                ctx.beginPath(); ctx.arc(0, -52, 11, 0, Math.PI * 2); ctx.fill();
                ctx.fillStyle = '#0f172a';
                ctx.fillRect(2, -55, 7, 5);

                // Headlamp light cone
                ctx.fillStyle = 'rgba(254, 240, 138, 0.15)';
                ctx.beginPath();
                ctx.moveTo(7, -53);
                ctx.lineTo(160, -85);
                ctx.lineTo(160, -20);
                ctx.closePath();
                ctx.fill();

                // Gun Model
                ctx.fillStyle = '#475569';
                ctx.fillRect(4, -38, 24, 7);
                ctx.fillRect(8, -32, 5, 10);
                if (this.weapon.piercing) {
                    ctx.fillStyle = '#a855f7';
                    ctx.fillRect(22, -40, 9, 11);
                }

                ctx.restore();

                // Player Tag Overhead
                ctx.fillStyle = this.classData.color;
                ctx.font = 'bold 12px "Chakra Petch", sans-serif';
                ctx.textAlign = 'center';
                ctx.fillText(`P${this.slot + 1} ${this.classData.name}`, this.x, this.y - 70);

                if (this.isReloading) {
                    const pct = 1 - (this.reloadTimer / this.weapon.reloadTime);
                    ctx.strokeStyle = '#facc15';
                    ctx.lineWidth = 4;
                    ctx.beginPath();
                    ctx.arc(this.x, this.y - 82, 11, -Math.PI/2, -Math.PI/2 + (Math.PI * 2 * pct));
                    ctx.stroke();
                }
            }
        }

        class Zombie {
            constructor(type, spawnSide, hpMultiplier = 1.0) {
                this.type = type;
                this.x = spawnSide === 'left' ? -50 : canvas.width + 50;
                this.y = groundY;
                this.facing = spawnSide === 'left' ? 1 : -1;
                this.animFrame = Math.random() * 10;
                this.hitFlashTimer = 0;

                if (type === 'runner') {
                    this.speed = 3.6; this.hp = 50 * hpMultiplier; this.maxHp = this.hp; this.damage = 12;
                    this.size = 18; this.color = '#dc2626'; this.height = 44;
                } else if (type === 'armored') {
                    this.speed = 1.4; this.hp = 260 * hpMultiplier; this.maxHp = this.hp; this.damage = 22;
                    this.size = 28; this.color = '#334155'; this.height = 58; this.hasShield = true;
                } else if (type === 'boss') {
                    this.speed = 1.0; this.hp = 950 * hpMultiplier; this.maxHp = this.hp; this.damage = 50;
                    this.size = 42; this.color = '#7f1d1d'; this.height = 82;
                } else {
                    this.speed = 1.8; this.hp = 95 * hpMultiplier; this.maxHp = this.hp; this.damage = 16;
                    this.size = 22; this.color = '#15803d'; this.height = 48;
                }
            }

            update(dt) {
                if (this.hitFlashTimer > 0) this.hitFlashTimer -= dt;

                let targetX = canvas.width / 2;
                let minDist = 9999;

                barricades.forEach(b => {
                    if (b.hp > 0) {
                        const d = Math.abs(this.x - b.x);
                        if (d < minDist) { minDist = d; targetX = b.x; }
                    }
                });

                players.forEach(p => {
                    if (!p.isDead) {
                        const d = Math.abs(this.x - p.x);
                        if (d < minDist) { minDist = d; targetX = p.x; }
                    }
                });

                this.facing = targetX > this.x ? 1 : -1;
                this.x += this.facing * this.speed;
                this.animFrame += 0.16;

                if (this.type === 'boss') {
                    if (Math.sin(this.animFrame) > 0.8) triggerScreenShake(2, 0.05);
                }

                barricades.forEach(b => {
                    if (b.hp > 0 && Math.abs(this.x - b.x) < this.size + 18) {
                        b.takeDamage(this.damage * dt);
                        this.x -= this.facing * this.speed;
                    }
                });

                players.forEach(p => {
                    if (!p.isDead && Math.abs(this.x - p.x) < this.size + 12) {
                        p.takeDamage(this.damage * dt);
                    }
                });
            }

            takeDamage(dmg, shooterSlot) {
                this.hp -= dmg;
                this.hitFlashTimer = 0.1;
                bloodSplatters.push(new BloodSplatter(this.x, groundY));
                particles.push(new Particle(this.x, this.y - this.height/2, (Math.random()-0.5)*4, -Math.random()*4, 5, '#15803d', 0.4));
                floatingTexts.push(new FloatingText(this.x, this.y - this.height, `-${Math.round(dmg)}`, '#facc15', 0.6));

                if (this.hp <= 0) {
                    if (players[shooterSlot]) players[shooterSlot].addKill();
                    playSound('zombie');
                    return true;
                }
                return false;
            }

            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.scale(this.facing, 1);

                const armAngle = Math.sin(this.animFrame) * 0.35;

                ctx.strokeStyle = '#0f172a';
                ctx.lineWidth = this.type === 'boss' ? 8 : 4;
                ctx.beginPath(); ctx.moveTo(-5, -16); ctx.lineTo(-5 + Math.sin(armAngle)*10, 0); ctx.stroke();
                ctx.beginPath(); ctx.moveTo(5, -16); ctx.lineTo(5 - Math.sin(armAngle)*10, 0); ctx.stroke();

                ctx.fillStyle = this.hitFlashTimer > 0 ? '#ffffff' : this.color;
                ctx.fillRect(-12, -this.height + 16, 24, this.height - 28);

                ctx.beginPath(); ctx.arc(0, -this.height + 8, this.size * 0.45, 0, Math.PI * 2); ctx.fill();

                // Glowing eyes
                ctx.fillStyle = '#ef4444';
                ctx.shadowColor = '#ef4444';
                ctx.shadowBlur = 10;
                ctx.fillRect(3, -this.height + 5, 4, 4);
                ctx.shadowBlur = 0;

                ctx.strokeStyle = '#334155';
                ctx.lineWidth = this.type === 'boss' ? 7 : 4;
                ctx.beginPath(); ctx.moveTo(0, -this.height + 24); ctx.lineTo(22, -this.height + 24 + Math.sin(armAngle)*6); ctx.stroke();

                if (this.hasShield) {
                    ctx.fillStyle = '#475569';
                    ctx.fillRect(14, -this.height + 5, 8, this.height - 10);
                    ctx.strokeStyle = '#94a3b8';
                    ctx.strokeRect(14, -this.height + 5, 8, this.height - 10);
                }

                ctx.restore();

                if (this.hp < this.maxHp) {
                    const pct = Math.max(0, this.hp / this.maxHp);
                    ctx.fillStyle = 'rgba(15, 23, 42, 0.85)';
                    ctx.fillRect(this.x - 22, this.y - this.height - 14, 44, 6);
                    ctx.fillStyle = '#ef4444';
                    ctx.fillRect(this.x - 22, this.y - this.height - 14, 44 * pct, 6);
                }
            }
        }

        /* 1 BULLET = 1 ZOMBIE ACCURATE HIT ENGINE */
        class Bullet {
            constructor(x, y, vx, vy, damage, color, piercing = false, shooterSlot = 0) {
                this.x = x; this.y = y;
                this.vx = vx; this.vy = vy;
                this.damage = damage;
                this.color = color;
                this.piercing = piercing;
                this.shooterSlot = shooterSlot;
                this.alive = true;
            }

            update() {
                this.x += this.vx;
                this.y += this.vy;

                if (this.x < 0 || this.x > canvas.width || this.y < 0 || this.y > canvas.height) {
                    this.alive = false;
                    return;
                }

                for (let i = zombies.length - 1; i >= 0; i--) {
                    const z = zombies[i];
                    if (Math.abs(this.x - z.x) < z.size + 10 && Math.abs(this.y - (z.y - z.height/2)) < z.height/2) {
                        const isKilled = z.takeDamage(this.damage, this.shooterSlot);
                        if (isKilled) zombies.splice(i, 1);

                        if (!this.piercing) {
                            this.alive = false;
                            break;
                        }
                    }
                }
            }

            draw() {
                ctx.fillStyle = this.color;
                ctx.shadowColor = this.color;
                ctx.shadowBlur = 10;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.piercing ? 7 : 3.5, 0, Math.PI * 2);
                ctx.fill();
                ctx.shadowBlur = 0;
            }
        }

        class MuzzleFlash {
            constructor(x, y, facing, color) {
                this.x = x; this.y = y; this.facing = facing; this.color = color;
                this.life = 0.08;
            }
            update(dt) { this.life -= dt; }
            draw() {
                if (this.life <= 0) return;
                ctx.fillStyle = this.color;
                ctx.shadowColor = this.color;
                ctx.shadowBlur = 25;
                ctx.beginPath();
                ctx.arc(this.x, this.y, 16, 0, Math.PI * 2);
                ctx.fill();
                ctx.shadowBlur = 0;
            }
        }

        class Barricade {
            constructor(x, y) {
                this.x = x; this.y = y;
                this.hp = 350; this.maxHp = 350;
            }
            takeDamage(dmg) { this.hp -= dmg; }
            draw() {
                if (this.hp <= 0) return;
                ctx.fillStyle = '#334155';
                ctx.fillRect(this.x - 22, this.y - 48, 44, 48);
                ctx.fillStyle = '#b45309';
                ctx.fillRect(this.x - 20, this.y - 36, 40, 12);
                
                const pct = Math.max(0, this.hp / this.maxHp);
                ctx.fillStyle = '#10b981';
                ctx.fillRect(this.x - 22, this.y - 58, 44 * pct, 5);
            }
        }

        class Grenade {
            constructor(x, y, vx, vy, shooterSlot) {
                this.x = x; this.y = y; this.vx = vx; this.vy = vy;
                this.shooterSlot = shooterSlot;
                this.timer = 1.1; this.alive = true;
            }
            update(dt) {
                this.x += this.vx; this.y += this.vy; this.vy += 0.45;
                if (this.y >= groundY - 12) { this.y = groundY - 12; this.vx *= 0.5; }
                this.timer -= dt;
                if (this.timer <= 0) this.explode();
            }
            explode() {
                this.alive = false;
                playSound('explosion');
                triggerScreenShake(14, 0.35);
                particles.push(new MuzzleFlash(this.x, this.y, 1, '#f97316'));

                for (let i = zombies.length - 1; i >= 0; i--) {
                    const z = zombies[i];
                    if (Math.abs(this.x - z.x) < 160) {
                        if (z.takeDamage(280, this.shooterSlot)) zombies.splice(i, 1);
                    }
                }
            }
            draw() {
                ctx.fillStyle = '#15803d';
                ctx.beginPath(); ctx.arc(this.x, this.y, 6, 0, Math.PI * 2); ctx.fill();
            }
        }

        class LaserBeam {
            constructor(x, y, facing, shooterSlot) {
                this.x = x; this.y = y; this.facing = facing; this.shooterSlot = shooterSlot;
                this.timer = 0.5; this.alive = true;

                for (let i = zombies.length - 1; i >= 0; i--) {
                    const z = zombies[i];
                    if ((facing === 1 && z.x >= x) || (facing === -1 && z.x <= x)) {
                        if (z.takeDamage(600, shooterSlot)) zombies.splice(i, 1);
                    }
                }
            }
            update(dt) {
                this.timer -= dt;
                if (this.timer <= 0) this.alive = false;
            }
            draw() {
                ctx.strokeStyle = '#a855f7';
                ctx.lineWidth = 22;
                ctx.shadowColor = '#c084fc';
                ctx.shadowBlur = 30;
                ctx.beginPath();
                ctx.moveTo(this.x, this.y);
                ctx.lineTo(this.facing === 1 ? canvas.width : 0, this.y);
                ctx.stroke();
                ctx.shadowBlur = 0;
            }
        }

        class Particle {
            constructor(x, y, vx, vy, size, color, life) {
                this.x = x; this.y = y; this.vx = vx; this.vy = vy;
                this.size = size; this.color = color; this.life = life; this.maxLife = life;
            }
            update(dt) { this.x += this.vx; this.y += this.vy; this.life -= dt; }
            draw() {
                ctx.globalAlpha = Math.max(0, this.life / this.maxLife);
                ctx.fillStyle = this.color;
                ctx.beginPath(); ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2); ctx.fill();
                ctx.globalAlpha = 1.0;
            }
        }

        class ShellCasing {
            constructor(x, y, vx) {
                this.x = x; this.y = y; this.vx = vx; this.vy = -3.5;
                this.life = 1.6;
            }
            update(dt) {
                this.x += this.vx; this.y += this.vy; this.vy += 0.35;
                if (this.y >= groundY) { this.y = groundY; this.vx = 0; }
                this.life -= dt;
            }
            draw() {
                if (this.life <= 0) return;
                ctx.fillStyle = '#facc15';
                ctx.fillRect(this.x, this.y, 4, 2);
            }
        }

        class BloodSplatter {
            constructor(x, y) {
                this.x = x + (Math.random() - 0.5) * 24;
                this.y = y;
                this.size = 8 + Math.random() * 14;
            }
            draw() {
                ctx.fillStyle = '#14532d';
                ctx.beginPath(); ctx.ellipse(this.x, this.y, this.size, this.size * 0.35, 0, 0, Math.PI * 2); ctx.fill();
            }
        }

        class FloatingText {
            constructor(x, y, text, color, life = 1.0) {
                this.x = x; this.y = y; this.text = text; this.color = color;
                this.life = life; this.maxLife = life;
            }
            update(dt) { this.y -= 18 * dt; this.life -= dt; }
            draw() {
                ctx.globalAlpha = Math.max(0, this.life / this.maxLife);
                ctx.fillStyle = this.color;
                ctx.font = 'bold 14px "Chakra Petch", sans-serif';
                ctx.textAlign = 'center';
                ctx.fillText(this.text, this.x, this.y);
                ctx.globalAlpha = 1.0;
            }
        }

        /* ----------------------------------------------------
           6. DYNAMIC WAVE DIRECTOR (SCALES WITH ROUND & PLAYERS)
        ---------------------------------------------------- */
        function initGamePlayers(count = 1, singleClass = 0) {
            players.length = 0;
            for (let i = 0; i < count; i++) {
                addPlayer(i, i === 0 && currentMode === 'single' ? singleClass : i);
            }
            currentRound = 1;
            startRound(1);
        }

        function addPlayer(slot, classIndex = slot) {
            if (players[slot]) return;
            players[slot] = new Player(slot, classIndex);
            updateHUDPlayerBadges();
        }

        function startRound(roundNum) {
            currentRound = roundNum;
            roundState = 'active';

            const activePlayerCount = Math.max(1, players.filter(p => p.connected).length);
            zombiesToSpawn = (10 + roundNum * 8) * activePlayerCount;

            spawnTimer = 0;

            document.getElementById('hud-round-text').innerText = `RAUND ${currentRound} / ${TOTAL_ROUNDS}`;
            showBanner(`RAUND ${currentRound}`, `Zombilar soni: ${zombiesToSpawn} ta (${activePlayerCount} o'yinchi)`);

            players.forEach(p => {
                if (!p.isDead) p.hp = Math.min(p.maxHp, p.hp + 35);
            });
        }

        function updateRoundDirector(dt) {
            if (roundState !== 'active') return;

            const activePlayerCount = Math.max(1, players.filter(p => p.connected).length);

            if (zombiesToSpawn > 0) {
                spawnTimer -= dt;
                if (spawnTimer <= 0) {
                    spawnTimer = Math.max(0.25, (1.8 - (currentRound * 0.12)) / Math.sqrt(activePlayerCount));
                    zombiesToSpawn--;

                    const side = Math.random() > 0.5 ? 'left' : 'right';
                    let type = 'walker';
                    const rand = Math.random();

                    if (currentRound >= 2 && rand < 0.32) type = 'runner';
                    else if (currentRound >= 4 && rand < 0.52) type = 'armored';
                    else if (currentRound >= 6 && rand > 0.80) type = 'boss';

                    const hpScaling = 1.0 + (currentRound - 1) * 0.2 + (activePlayerCount - 1) * 0.25;
                    zombies.push(new Zombie(type, side, hpScaling));
                }
            }

            if (zombiesToSpawn <= 0 && zombies.length === 0) {
                roundState = 'cleared';
                if (currentRound < TOTAL_ROUNDS) {
                    showBanner("RAUND YAKUNLANDI!", "Keyingi raundga tayyorlaning...");
                    setTimeout(() => startRound(currentRound + 1), 3500);
                } else {
                    triggerVictory();
                }
            }

            const alivePlayers = players.filter(p => !p.isDead);
            if (players.length > 0 && alivePlayers.length === 0) {
                triggerGameOver();
            }
        }

            document.getElementById('banner-sub').innerText = sub;
            banner.classList.remove('opacity-0', 'translate-y-4');
            setTimeout(() => {
                banner.classList.add('opacity-0', 'translate-y-4');
            }, 2800);
        }

        function restartGame() {
            document.getElementById('game-over-modal').classList.add('hidden');
            zombies.length = 0;
            bullets.length = 0;
            barricades.length = 0;
            grenades.length = 0;
            lasers.length = 0;

        /* ----------------------------------------------------
           7. MULTI-LAYERED BACKGROUND & RENDERER
        ---------------------------------------------------- */
        function initCanvasGame() {
            canvas = document.getElementById('game-canvas');
            ctx = canvas.getContext('2d', { alpha: false });

            window.addEventListener('resize', resizeCanvas);
            resizeCanvas();

            for (let i = 0; i < 25; i++) {
                fogParticles.push({
                    x: Math.random() * window.innerWidth,
                    y: window.innerHeight - 150 + Math.random() * 100,
                    radius: 60 + Math.random() * 90,
                    vx: 0.2 + Math.random() * 0.4
                });
            }

            setupKeyboardSinglePlayer();
            requestAnimationFrame(gameLoop);
        }

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
            groundY = canvas.height - 85;
        }

        function setupKeyboardSinglePlayer() {
            window.addEventListener('keydown', (e) => {
                initAudio();
                if (currentMode !== 'single') return;
                const p = players[0];
                if (!p) return;

                if (e.code === 'KeyA' || e.code === 'ArrowLeft') p.inputLeft = true;
                if (e.code === 'KeyD' || e.code === 'ArrowRight') p.inputRight = true;
                if (e.code === 'Space' || e.code === 'KeyJ') p.inputShoot = true;
                if (e.code === 'KeyR') p.reloadWeapon();
                if (e.code === 'KeyE' || e.code === 'ShiftLeft') p.useAbility();
            });

            window.addEventListener('keyup', (e) => {
                if (currentMode !== 'single') return;
                const p = players[0];
                if (!p) return;

                if (e.code === 'KeyA' || e.code === 'ArrowLeft') p.inputLeft = false;
                if (e.code === 'KeyD' || e.code === 'ArrowRight') p.inputRight = false;
                if (e.code === 'Space' || e.code === 'KeyJ') p.inputShoot = false;
            });
        }

        let lastTime = performance.now();

        function gameLoop(now) {
            requestAnimationFrame(gameLoop);

            const dt = Math.min((now - lastTime) / 1000, 0.1);
            lastTime = now;

            if (currentMode !== 'joystick') {
                updateRoundDirector(dt);

                players.forEach(p => p.update(dt));
                zombies.forEach(z => z.update(dt));

                for (let i = bullets.length - 1; i >= 0; i--) {
                    bullets[i].update();
                    if (!bullets[i].alive) bullets.splice(i, 1);
                }

                for (let i = grenades.length - 1; i >= 0; i--) {
                    grenades[i].update(dt);
                    if (!grenades[i].alive) grenades.splice(i, 1);
                }

                for (let i = lasers.length - 1; i >= 0; i--) {
                    lasers[i].update(dt);
                    if (!lasers[i].alive) lasers.splice(i, 1);
                }

                for (let i = particles.length - 1; i >= 0; i--) {
                    particles[i].update(dt);
                    if (particles[i].life <= 0) particles.splice(i, 1);
                }

                for (let i = floatingTexts.length - 1; i >= 0; i--) {
                    floatingTexts[i].update(dt);
                    if (floatingTexts[i].life <= 0) floatingTexts.splice(i, 1);
                }

                fogParticles.forEach(f => {
                    f.x += f.vx;
                    if (f.x > canvas.width + 100) f.x = -100;
                });

                syncStateToControllers();
                renderCanvas();
            }
        }

        function renderCanvas() {
            const w = canvas.width;
            const h = canvas.height;

            ctx.save();

            if (screenShakeTime > 0) {
                screenShakeTime -= 0.016;
                const rx = (Math.random() - 0.5) * screenShakeIntensity;
                const ry = (Math.random() - 0.5) * screenShakeIntensity;
                ctx.translate(rx, ry);
            }

            // Overcast Sky Background
            const skyGrad = ctx.createLinearGradient(0, 0, 0, h);
            skyGrad.addColorStop(0, '#090d16');
            skyGrad.addColorStop(0.5, '#1e293b');
            skyGrad.addColorStop(1, '#334155');
            ctx.fillStyle = skyGrad;
            ctx.fillRect(0, 0, w, h);

            // Classical Government Building Silhouette
            const centerW = Math.min(500, w * 0.5);
            const centerLeft = (w - centerW) / 2;
            ctx.fillStyle = '#1e293b';
            ctx.fillRect(centerLeft, groundY - 240, centerW, 240);
            
            // Pillars
            ctx.fillStyle = '#334155';
            const pillarCount = 8;
            const pStep = centerW / (pillarCount + 1);
            for (let p = 1; p <= pillarCount; p++) {
                ctx.fillRect(centerLeft + p * pStep - 6, groundY - 220, 12, 220);
            }
            // Roof Pediment
            ctx.beginPath();
            ctx.moveTo(centerLeft - 20, groundY - 240);
            ctx.lineTo(w / 2, groundY - 310);
            ctx.lineTo(centerLeft + centerW + 20, groundY - 240);
            ctx.closePath();
            ctx.fill();

            // Destroyed Buildings (Left & Right)
            ctx.fillStyle = '#0f172a';
            for (let i = 0; i < w; i += 110) {
                if (i < centerLeft - 40 || i > centerLeft + centerW + 40) {
                    const bHeight = 160 + Math.sin(i * 0.015) * 110;
                    ctx.fillRect(i, groundY - bHeight, 95, bHeight);
                }
            }

            // Streetlamp Light Cone
            ctx.fillStyle = 'rgba(254, 240, 138, 0.06)';
            ctx.beginPath();
            ctx.moveTo(180, groundY - 180);
            ctx.lineTo(80, groundY);
            ctx.lineTo(280, groundY);
            ctx.fill();

            // Asphalt Street & Markings
            ctx.fillStyle = '#090d16';
            ctx.fillRect(0, groundY, w, h - groundY);

            ctx.strokeStyle = '#ca8a04';
            ctx.lineWidth = 5;
            ctx.setLineDash([30, 25]);
            ctx.beginPath();
            ctx.moveTo(0, groundY + 40);
            ctx.lineTo(w, groundY + 40);
            ctx.stroke();
            ctx.setLineDash([]);

            bloodSplatters.forEach(b => b.draw());
            barricades.forEach(b => b.draw());
            zombies.forEach(z => z.draw());
            players.forEach(p => p.draw());
            bullets.forEach(b => b.draw());
            grenades.forEach(g => g.draw());
            lasers.forEach(l => l.draw());
            particles.forEach(pt => pt.draw());
            floatingTexts.forEach(ft => ft.draw());

            // Volumetric Fog Layer
            ctx.fillStyle = 'rgba(30, 41, 59, 0.12)';
            fogParticles.forEach(f => {
                ctx.beginPath(); ctx.arc(f.x, f.y, f.radius, 0, Math.PI * 2); ctx.fill();
            });

            ctx.restore();

            updateHUDPlayerBadges();
        }

                                <span class="text-amber-400 ml-1">🎯 ${p.kills}</span>
                            </div>
                            <div class="w-14 h-1 bg-slate-900 rounded-full overflow-hidden mt-0.5 border border-slate-700">
                                <div class="h-full bg-emerald-400" style="width: ${hpPct}%"></div>
                            </div>
                        </div>
                    </div>
